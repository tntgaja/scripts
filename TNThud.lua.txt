-- Core Functions (UltraStuff method optimized)
local function getFocusedBall()
    local ballsFolder = Workspace:FindFirstChild("Balls")
    if not ballsFolder then return nil end
    for _, ball in pairs(ballsFolder:GetChildren()) do
        if ball:GetAttribute("target") then
            local targetId = ball:GetAttribute("target")
            local targetPlayer = Players:GetPlayerByUserId(tonumber(targetId) or 0)
            if targetPlayer == player then
                return ball
            end
        elseif ball:FindFirstChild("realBall") then  -- Backup detect
            return ball
        end
    end
    return nil
end

local function startAutoParry()
    heartbeatCon = RunService.Heartbeat:Connect(function()
        local ball = getFocusedBall()
        if not ball then return end
        
        local char = player.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end
        local hrp = char.HumanoidRootPart
        local dist = (ball.Position - hrp.Position).Magnitude
        
        if dist > getgenv().BladeConfig.Distance then return end
        
        local ballVel = ball.AssemblyLinearVelocity.Magnitude
        local timeToImpact = dist / math.max(ballVel, 1)
        local dynamicThreshold = math.clamp(0.12 + (30 / ballVel), 0.08, 0.25)  -- Pro timing: fast ball = tight parry
        
        if timeToImpact <= dynamicThreshold then
            if getgenv().BladeConfig.UseRage then
                pcall(function() AbilityRemote:FireServer() end)
                task.wait(0.01)
            end
            pcall(function() ParryRemote:FireServer() end)
        end
    end)
end

local function stopAutoParry()
    if heartbeatCon then
        heartbeatCon:Disconnect()
        heartbeatCon = nil
    end
end

local function toggleAutoSpam()
    if getgenv().BladeConfig.AutoSpam then
        spawn(function()
            while getgenv().BladeConfig.AutoSpam do
                pcall(function() ParryRemote:FireServer() end)
                task.wait(0.08)  -- Spam rate anti-detect
            end
        end)
    end
end
